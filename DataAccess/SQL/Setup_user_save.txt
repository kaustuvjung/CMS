SET XACT_ABORT ON
BEGIN TRANSACTION trn_user

BEGIN TRY
	--IF NOT EXISTS(SELECT  TOP 1 1  FROM OPENJSON)
	IF @id = 0
		BEGIN
			INSERT INTO dbo.User([Name], Email, MobileNumber, UserName, [Password], Salt, DocumentDataId, IsStatic, IsAdmin, IsActive, CreatedBy, CreatedDate)
					VALUES(@name , @email, @mobilenumber , @username, @password, @salt, NEWID(), 0,0,1,@crearedBy, GETDATE() );

			DECLARE @newId as INT = SCOPE_IDENTITY();

			INSERT INTO dbo.UserPermission(UserId, PermissionId)
			      VALUES(@newId, @permissionId);

			SET @id =@newId 
		END
	ELSE
		BEGIN
			Update dbo.User SET
				[Name] = @name,
				UserName =  @username,
				Email = @email,
				MobileNumber =  @mobilenumber,
				ModifiedBy = @modifiedBy,
				ModifiedDate =  GetDate()
			WHERE Id = @id;
			DELETE FROM dbo.UserPermission Where UserId =  @id
			INSER INTO dbo.UserPermission(UserId, PermissionId)
				VALUES(@id, @permissionId);
		END
COMMIT
END TRY

BEGIN CATCH 
	ROLLBACK
	DECLARE @error AS NVARCHAR(MAX) = ERROR_MESSAGE();
	THROW 51000, @error , 1
END CATCH
SELECT @Id AS ID
